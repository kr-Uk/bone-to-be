# BoneToBe 실행 가이드

## 📌 프로젝트 소개
BoneToBe는 사용자의 체형 분석을 기반으로 의류 가상 피팅 추천을 제공하는 서비스입니다.

AI 기반 체형 분석과 가상 피팅을 통해 사용자 맞춤 스타일을 제공합니다.

---

## 🖥 개발 환경

### Backend
- Java 17
- Spring Boot 3.6.0
- Gradle 8.7
- JPA / Hibernate

### Frontend
- React 18
- Node 22
- Vite

### Infrastructure
- Docker Compose
- PostgreSQL 15
- Redis
- RabbitMQ
- Nginx

### AI Worker
- Python
- Shapy Model
- Gemini API

3️⃣ Docker 실행
docker-compose.yml
```
version: "3.9"

networks:
  bonetobe-net:
    driver: bridge

volumes:
  postgres-data:
  rabbitmq-data:
  grafana-data:
  prometheus-data:
  loki-data:
  promtail-positions:

services:
  postgres:
    image: postgres:${POSTGRES_VERSION}
    container_name: bonetobe-postgres
    restart: always
    environment:
      POSTGRES_DB: ${POSTGRES_DB}
      POSTGRES_USER: ${POSTGRES_USER}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD}
    volumes:
      - postgres-data:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ${POSTGRES_USER} -d ${POSTGRES_DB}"]
      interval: 5s
      timeout: 3s
      retries: 20
    networks:
      - bonetobe-net

  redis:
    image: redis:${REDIS_VERSION}
    container_name: bonetobe-redis
    restart: always
    command: ["redis-server", "--appendonly", "yes"]
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 5s
      timeout: 3s
      retries: 20
    networks:
      - bonetobe-net

  rabbitmq:
    image: rabbitmq:${RABBITMQ_VERSION}
    container_name: bonetobe-rabbitmq
    restart: always
    environment:
      RABBITMQ_DEFAULT_USER: ${RABBITMQ_USERNAME:-guest}
      RABBITMQ_DEFAULT_PASS: ${RABBITMQ_PASSWORD:-guest}
      RABBITMQ_DEFAULT_VHOST: ${RABBITMQ_VHOST:-/}
    ports:
      - "${RABBITMQ_MGMT_PORT}:15672"
      - "5672:5672"
    healthcheck:
      test: ["CMD-SHELL", "rabbitmq-diagnostics -q ping"]
      interval: 5s
      timeout: 5s
      retries: 20
    volumes:
      - rabbitmq-data:/var/lib/rabbitmq/data
    networks:
      - bonetobe-net

  backend:
    image: ${BACKEND_IMAGE:-taekminkwon/backend:latest}
    container_name: bonetobe-backend
    restart: always
    env_file:
      - .env
    depends_on:
      postgres:
        condition: service_healthy
      redis:
        condition: service_healthy
      rabbitmq:
        condition: service_healthy
    ports:
      - "8080:8080"
    networks:
      - bonetobe-net
    environment:
      JAVA_TOOL_OPTIONS:
        -Djava.net.preferIPV4Stack=true

  frontend:
    image: ${FRONTEND_IMAGE:-taekminkwon/frontend:latest}
    container_name: bonetobe-frontend
    restart: always
    env_file:
      - .env
    depends_on:
      - backend
    ports:
      - "5173:80"
      - "5174:80"
    networks:
      - bonetobe-net

  prometheus:
    image: prom/prometheus
    container_name: bonetobe-prometheus
    restart: unless-stopped
    volumes:
      - ./prometheus/conf/prometheus.yml:/etc/prometheus/prometheus.yml:ro
      - prometheus-data:/prometheus
    command:
      - "--config.file=/etc/prometheus/prometheus.yml"
    networks:
      - bonetobe-net

  grafana:
    image: grafana/grafana
    container_name: bonetobe-grafana
    restart: unless-stopped
    depends_on:
      - prometheus
    ports:
      - "127.0.0.1:3000:3000"
    environment:
      - GF_SERVER_ROOT_URL=http://localhost:3000/
      - GF_SERVER_SERVE_FROM_SUB_PATH=false
      - GF_SECURITY_ADMIN_USER=${GF_ADMIN_USER}
      - GF_SECURITY_ADMIN_PASSWORD=${GF_ADMIN_PASSWORD}
      - GF_USERS_ALLOW_SIGN_UP=false
      - GF_AUTH_ANONYMOUS_ENABLED=false
    volumes:
      - grafana-data:/var/lib/grafana
    networks:
      - bonetobe-net
  loki:
    image: grafana/loki:2.9.0
    container_name: bonetobe-loki
    restart: unless-stopped
    command: -config.file=/etc/loki/config.yml
    volumes:
      - ./loki/config.yml:/etc/loki/config.yml:ro
      - loki-data:/loki
    ports:
      - "127.0.0.1:3100:3100"
    networks:
      - bonetobe-net

  promtail:
    image: grafana/promtail:2.9.0
    container_name: bonetobe-promtail
    restart: unless-stopped
    command: -config.file=/etc/promtail/config.yml
    volumes:
      - ./promtail/config.yml:/etc/promtail/config.yml:ro
      - /var/lib/docker/containers:/var/lib/docker/containers:ro
      - /var/run/docker.sock:/var/run/docker.sock:ro
      - promtail-positions:/tmp
    depends_on:
      - loki
    networks:
      - bonetobe-net
```
docker-compose up -d

server {
    server_name bonetobe.site www.bonetobe.site;

    # frontend
    location / {
        proxy_pass http://127.0.0.1:5173;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # backend
    location ^~ /api/ {
        proxy_pass http://127.0.0.1:8080;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        proxy_buffering off;
        proxy_cache off;

        proxy_read_timeout 3600;
        proxy_send_timeout 3600;
    }

     # grafana
    location = /grafana {return 404;}
    location ^~ /grafana/ {return 404;}


    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/bonetobe.site/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/bonetobe.site/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot


}

server {
    if ($host = www.bonetobe.site) {
        return 301 https://$host$request_uri;
    } # managed by Certbot


    if ($host = bonetobe.site) {
        return 301 https://$host$request_uri;
    } # managed by Certbot


    listen 80;
    server_name bonetobe.site www.bonetobe.site;
    return 404; # managed by Certbot
}
